/**
 * Shop orders - Google Sheets backend
 * -----------------------------------
 * Used by the Shop tab on samisdiving.com (index.html): mask strap + fins strap.
 * Tab "Shop orders": one row per order, newest at the bottom (Status starts as "New").
 *
 *  - GET                                      -> { ok:true }  (health check)
 *  - POST {action:"order", name, phone, items:[{product,qty,colors}], address, kecamatan, city, province, postcode, note}
 *                                             -> { ok, order:{id,subtotal,shipping,total,courier} }
 *    Prices and the J&T shipping fee are always re-calculated here, never trusted from the website.
 *
 * SET UP (once)
 *  1. Create a new Google Sheet > Extensions > Apps Script > paste this whole file.
 *  2. Deploy > New deployment > Web app > Execute as: Me > Who has access: Anyone. Authorize when asked.
 *  3. Send the web app link (ends in /exec) to Claude to put into index.html (SHOP_API).
 * After editing this file: Deploy > Manage deployments > Edit > New version.
 *
 * The J&T fees (FEES below) must match the FEES list in index.html. Tell Claude when they change and both are updated.
 */
const ORDERS_SHEET = 'Shop orders';
const TIMEZONE = 'Asia/Makassar';

// Prices in Rp, weight in grams.
const PRODUCTS = {
  'mask strap': { name: 'Mask strap', price: 100000, weight: 100 },
  'fins strap': { name: 'Fins strap', price: 200000, weight: 250 }
};

// J&T fee for a 1 kg parcel from Abang, per province, in Rp. A province that is not listed has no fixed fee:
// the order is saved without shipping and Sam confirms the fee on WhatsApp. Heavier orders are charged per started kg.
const FEES = {
  // 'Bali': 12000,
};

const PROVINCES = ['Aceh', 'Sumatera Utara', 'Sumatera Barat', 'Riau', 'Kepulauan Riau', 'Jambi', 'Sumatera Selatan', 'Bangka Belitung', 'Bengkulu', 'Lampung',
  'DKI Jakarta', 'Banten', 'Jawa Barat', 'Jawa Tengah', 'DI Yogyakarta', 'Jawa Timur', 'Bali', 'Nusa Tenggara Barat', 'Nusa Tenggara Timur',
  'Kalimantan Barat', 'Kalimantan Tengah', 'Kalimantan Selatan', 'Kalimantan Timur', 'Kalimantan Utara',
  'Sulawesi Utara', 'Gorontalo', 'Sulawesi Tengah', 'Sulawesi Barat', 'Sulawesi Selatan', 'Sulawesi Tenggara',
  'Maluku', 'Maluku Utara', 'Papua', 'Papua Barat', 'Papua Barat Daya', 'Papua Selatan', 'Papua Tengah', 'Papua Pegunungan'];

const HEADERS = ['Order ID', 'Time', 'Status', 'Name', 'WhatsApp', 'Items', 'Subtotal (Rp)', 'Shipping (Rp)', 'Total (Rp)',
  'Full address', 'Kecamatan', 'Kota/Kabupaten', 'Province', 'Postcode', 'Note'];

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function errText_(err) { return String(err && err.message ? err.message : err); }

// Stops text from being read as a formula when it lands in the sheet.
function clean_(v, max) {
  let s = String(v == null ? '' : v).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').trim().slice(0, max || 300);
  if (/^[=+\-@]/.test(s)) s = "'" + s;
  return s;
}

// Checks the cart lines against the price list and works out subtotal + weight.
function cart_(items) {
  if (!Array.isArray(items) || !items.length || items.length > 20) throw new Error('Your cart is empty');
  let subtotal = 0, weight = 0;
  const lines = items.map(function (it) {
    const p = PRODUCTS[String(it.product)];
    const qty = Math.floor(Number(it.qty));
    if (!p || !(qty >= 1 && qty <= 20)) throw new Error('Unknown product in the cart');
    subtotal += p.price * qty;
    weight += p.weight * qty;
    return qty + 'x ' + p.name + (it.colors ? ' - ' + clean_(it.colors, 200) : '');
  });
  return { subtotal: subtotal, weight: weight, lines: lines };
}

function province_(name) {
  const want = String(name || '').trim().toLowerCase();
  return PROVINCES.filter(function (p) { return p.toLowerCase() === want; })[0] || '';
}

// null = no fixed fee for this province (Sam confirms it on WhatsApp)
function shipping_(province, weight) {
  const fee = Number(FEES[province]);
  return fee > 0 ? fee * Math.max(1, Math.ceil(weight / 1000)) : null;
}

function doGet() {
  return jsonOut_({ ok: true, message: 'Shop backend is running' });
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    if (body.action === 'order') return jsonOut_(placeOrder_(body));
    return jsonOut_({ ok: false, error: 'Unknown action' });
  } catch (err) {
    return jsonOut_({ ok: false, error: errText_(err) });
  }
}

function placeOrder_(b) {
  const name = clean_(b.name, 80), phone = String(b.phone || '').replace(/\D/g, '');
  if (name.length < 2) throw new Error('Please enter your name');
  if (phone.length < 9 || phone.length > 15) throw new Error('Please enter a valid WhatsApp number');
  const address = clean_(b.address, 300), kecamatan = clean_(b.kecamatan, 80), city = clean_(b.city, 80),
    province = province_(b.province), postcode = String(b.postcode || '').replace(/\D/g, '');
  if (address.length < 5) throw new Error('Please enter your full address');
  if (kecamatan.length < 2) throw new Error('Please enter your kecamatan');
  if (city.length < 2) throw new Error('Please enter your kota / kabupaten');
  if (!province) throw new Error('Please choose your province');
  if (postcode.length !== 5) throw new Error('Please enter your 5-digit postcode');

  const c = cart_(b.items);
  const shipping = shipping_(province, c.weight);
  const total = c.subtotal + (shipping || 0);

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(ORDERS_SHEET) || ss.insertSheet(ORDERS_SHEET);
    if (sh.getLastRow() === 0) {
      sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
      sh.setFrozenRows(1);
    }
    const day = Utilities.formatDate(new Date(), TIMEZONE, 'yyMMdd');
    const ids = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues() : [];
    const n = ids.filter(function (r) { return String(r[0]).indexOf('S' + day + '-') === 0; }).length + 1;
    const id = 'S' + day + '-' + (n < 10 ? '00' : n < 100 ? '0' : '') + n;
    const row = sh.getLastRow() + 1;
    sh.getRange(row, 1, 1, HEADERS.length).setNumberFormat('@');
    sh.getRange(row, 1, 1, HEADERS.length).setValues([[
      id, Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd HH:mm'), 'New', name, '+' + phone,
      c.lines.join('\n'), c.subtotal, shipping == null ? 'to confirm' : shipping, total,
      address, kecamatan, city, province, postcode, clean_(b.note, 300)
    ]]);
    sh.getRange(row, 7, 1, 3).setNumberFormat('#,##0');
    return { ok: true, order: { id: id, subtotal: c.subtotal, shipping: shipping, total: total, courier: 'J&T' } };
  } finally {
    lock.releaseLock();
  }
}
